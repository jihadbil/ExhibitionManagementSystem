static Exhibition NewExhibition(int tid, int venueId, string name) => new()
{ TenantID = tid, VenueID = venueId, Name = name, EntryCurrency = "LYD", StartDate = DateTime.UtcNow.Date, EndDate = DateTime.UtcNow.Date.AddDays(2) };

static async Task Probe(string name, Func<DbContextOptions<ApplicationDbContext>, int, int, Task<object>> run)
{
    var selectedProbes = Environment.GetEnvironmentVariable("EMS_AUDIT_PROBES");
    if (!string.IsNullOrWhiteSpace(selectedProbes) && !selectedProbes.Split(',').Contains(name)) return;
    var databaseName = "EMS_Review_20261017_" + Guid.NewGuid().ToString("N");
    var conn = new SqlConnectionStringBuilder("Server=localhost;Database=master;Integrated Security=True;TrustServerCertificate=True;Connect Timeout=5")
    { InitialCatalog = databaseName };
    var options = new DbContextOptionsBuilder<ApplicationDbContext>().UseSqlServer(conn.ConnectionString).Options;
    await using var seed = new ApplicationDbContext(options, new WorkflowFixture.BypassContext());
    try
    {
        await seed.Database.MigrateAsync();
        var invoice = await WorkflowFixture.SeedAsync(seed, InvoiceStatus.Draft);
        var exhibition = await seed.Exhibitions.SingleAsync();
        exhibition.StartDate = DateTime.UtcNow.Date; exhibition.EndDate = DateTime.UtcNow.Date.AddDays(2);
        await seed.SaveChangesAsync();
        var observation = await run(options, invoice.TenantID, invoice.InvoiceID);
        Console.WriteLine(JsonSerializer.Serialize(new { Scenario = name, observation }));
    }
    catch (Exception ex)
    {
        Console.WriteLine(JsonSerializer.Serialize(new { Scenario = name, HarnessError = ex.GetType().Name, ex.Message }));
    }
    finally
    {
        await seed.Database.EnsureDeletedAsync();
        Console.WriteLine($"Cleanup: {databaseName} deleted");
    }
}

sealed class TenantScope(int tenantId) : ITenantContext
{
    public int? TenantId => tenantId;
    public bool CanBypassTenantIsolation => false;
}
